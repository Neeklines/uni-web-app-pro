-- =========================================
-- DATABASE SCHEMA
-- SQLAlchemy → PostgreSQL/MySQL Compatible
-- =========================================

CREATE TABLE users (
    id INTEGER PRIMARY KEY,

    email VARCHAR NOT NULL UNIQUE,
    password VARCHAR NOT NULL,

    -- profile
    display_name VARCHAR NOT NULL DEFAULT '',

    -- preferences
    theme VARCHAR NOT NULL DEFAULT 'dark',
    currency VARCHAR NOT NULL DEFAULT 'PLN',
    date_format VARCHAR NOT NULL DEFAULT 'dd.MM.yyyy',

    -- notifications
    in_app_notifications BOOLEAN NOT NULL DEFAULT TRUE,
    email_notifications BOOLEAN NOT NULL DEFAULT TRUE
);


CREATE TABLE subscriptions (
    id INTEGER PRIMARY KEY,

    name VARCHAR NOT NULL,
    price FLOAT NOT NULL,

    billing_cycle VARCHAR NOT NULL,
    next_payment_date DATE NOT NULL,

    category VARCHAR NULL,

    is_pinned BOOLEAN NOT NULL DEFAULT FALSE,
    is_favourite BOOLEAN NOT NULL DEFAULT FALSE,
    is_active BOOLEAN NOT NULL DEFAULT TRUE,

    cancelled_at DATE NULL,
    notes TEXT NULL,

    user_id INTEGER NOT NULL,

    CONSTRAINT fk_subscription_user
        FOREIGN KEY (user_id)
        REFERENCES users(id)
        ON DELETE CASCADE
);

-- =========================================
-- INDEXES
-- =========================================

CREATE UNIQUE INDEX ix_users_email
ON users(email);

CREATE INDEX ix_subscriptions_id
ON subscriptions(id);

-- =========================================
-- RELATIONSHIPS
-- =========================================
-- USER 1 ----< N SUBSCRIPTION
--
-- One USER can have many SUBSCRIPTIONS
-- One SUBSCRIPTION belongs to exactly one USER
-- =========================================
